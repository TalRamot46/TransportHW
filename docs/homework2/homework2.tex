\section{Question 1 - Useful identity for the solution of the time-dependent slab-geometry Boltzmann equation}
In this question we are asked to prove the following identity:
\begin{equation}
    \psi(x,\mu,t;c) = ce^{-(1-c)t}\psi(cx,\mu,ct;1) \label{eq:identity}
\end{equation}
We will do it in two parts. First, we will write down the equation that $\tilde{\psi} = \psi(cx,\mu,ct;1)$ satisfies, and then we will use it in order to prove that the product $ce^{-(1-c)t}\tilde{\psi}$ satisfies the same equation as $\psi(x,\mu,t;c)$.
\subsection{The equation for $\tilde{\psi}$}
\paragraph{Notation:} We will denote $\frac{\partial \tilde{\psi}(x,\mu,t)}{\partial \tilde{x}}$ as the full derivative (before applying the chain rule) and $\frac{\partial \tilde{\psi}(x,\mu,t)}{\partial x}$ as the partial derivative (after applying the chain rule, and assuming that the $cx$ and $ct$ are parameters \textit{substituted} into the derivative (and not those that the derivative is taken with respect to)). We will denote the same for the time derivative.
First, from the definition of $\tilde{\psi}$ we have:
\begin{equation}
    \frac{1}{v}\frac{\partial \tilde{\psi}}{\partial \tilde{t}} + \mu \frac{\partial \tilde{\psi}}{\partial \tilde{x}} + \Sigma_t \tilde{\psi} = \frac{1}{2}\int_{-1}^{1} \tilde{\psi}d\mu^\prime
\end{equation}
which after applying the chain rule becomes:
\begin{equation}
    \frac{c}{v}\frac{\partial \tilde{\psi}}{\partial t} + c\mu \frac{\partial \tilde{\psi}}{\partial x} + \Sigma_t \tilde{\psi} = \frac{1}{2}\int_{-1}^{1} \tilde{\psi}d\mu^\prime
\end{equation}
\subsection{Proving the identity}
In this part we are going to simply subtitute the RHS of \ref{eq:identity} into the LHS of the equation for $\psi(x,\mu,t;c)$ and show that it satisfies the same equation. After substitution and application of the chain rule we get:
\begin{align}
    \text{LHS} &= \frac{1}{v}\left[c(c-1)e^{-(1-c)t}\tilde{\psi}+c^2 e^{-(1-c)t}\frac{\partial\tilde{\psi}}{\partial t}\right] + \mu c^2 e^{-(1-c)t}\frac{\partial\tilde{\psi}}{\partial x} + \Sigma_t c e^{-(1-c)t}\tilde{\psi} \\
    &= c e^{-(1-c)t}\left[\frac{c}{v}\frac{\partial\tilde{\psi}}{\partial t} + c\mu \frac{\partial\tilde{\psi}}{\partial x} + \Sigma_t \tilde{\psi} + \frac{c-1}{v}\tilde{\psi}\right] \\
    &= c e^{-(1-c)t}\left[\frac{1}{2}\int_{-1}^{1} \tilde{\psi}d\mu^\prime + \frac{c-1}{v}\tilde{\psi}\right] \\
\end{align}
\section{Question 2 - Derivation of the time-dependent planar scalar flux for a plane source}
The relation between a plane source and a point source as studied in class is given by:
\begin{equation}
    \phi_{\text{pl}}(x) = 2\pi\int_{|x|}^{\infty} \phi_{\text{pt}}(r)rdr
\end{equation}
The solution taken from J. C. J. Paasschens is the point-source scalar flux $\phi_{\text{pt}}(r,t)$ for $c=1$, and therefore the planar-source scalar flux is given by the integration given above.
\begin{align}
    \phi_{\text{pt}}(r,t) &= 2\pi \int_{|x|}^{\infty} \frac{e^{-v\Sigma_t t}}{{4\pi r^2}}\delta(r-vt)rdr + \int_{|x|}^{\infty} \frac{{\left(1-r^2/v^2t^2\right)}^{1/8}}{\left(4\pi vt/(3\Sigma_t)\right)^{3/2}} e^{-v\Sigma_t t}G\left(v\Sigma_t t\left[1-\frac{r^2}{v^2t^2}\right]^{3/4}\right)\Theta(vt-r)rdr \\
    &= \frac{e^{-v\Sigma_t t}}{2vt}\Theta(vt-|x|) + \text{sign}(vt-|x|) \frac{e^{-v\Sigma_t t}}{\left(4\pi vt/(3\Sigma_t)\right)^{3/2}} \int_{|x|}^{vt} \left(1-\frac{r^2}{v^2t^2}\right)^{1/8} G\left(v\Sigma_t t\left[1-\frac{r^2}{v^2t^2}\right]^{3/4}\right)rdr \\
    &= \frac{e^{-v\Sigma_t t}}{2vt}\Theta(vt-|x|) + \text{sign}(vt-|x|) \frac{e^{-v\Sigma_t t}(vt)^{1/2}}{\left(4\pi /(3\Sigma_t)\right)^{3/2}} \int_{\frac{|x|}{vt}}^{1} \left(1-\xi^2\right)^{1/8} G\left(v\Sigma_t t\left[1-\xi^2\right]^{3/4}\right) \xi d\xi
\end{align}
and then the integral can be evaluated numerically to get the planar-source scalar flux, and of course the relation proved in Q1 can be used to get the solution for $c\neq 1$ using a simple scaling and multiplication by an exponential factor of the solution for $c=1$.
